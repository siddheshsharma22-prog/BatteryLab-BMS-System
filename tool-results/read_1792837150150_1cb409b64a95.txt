    94→export function DesignTab() {
    95→  const { design, updateDesign, resetDesign, pack, sendCopilotMessage, copilotBusy } = useLabStore();
    96→  const { toast } = useToast();
    97→  const [saved, setSaved] = useState<SavedDesign[]>([]);
    98→  const [saving, setSaving] = useState(false);
    99→  const [pickId, setPickId] = useState<string>("");
   100→
   101→  const loadSaved = async () => {
   102→    try {
   103→      const res = await fetch("/api/designs");
   104→      const data = await res.json();
   105→      if (res.ok) setSaved(data.designs);
   106→    } catch {
   107→      /* silent */
   108→    }
   109→  };
   110→
   111→  useEffect(() => {
   112→    loadSaved();
   113→  }, []);
   114→
   115→  const issues = designIssues(design);
   116→  const complete = issues.length === 0;
   117→  const pm = packMath(design, pack);
   118→  const cellWh = design.capacityAh * (design.nominalV > 0 ? design.nominalV : (design.vmax + design.vmin) / 2 + 0.05);
   119→  const whKg = design.massKg > 0 ? cellWh / design.massKg : 0;
   120→
   121→  const save = async () => {
   122→    if (!complete) {
   123→      toast({ title: "Design incomplete", description: "Fill every required field before saving.", variant: "destructive" });
   124→      return;
   125→    }
   126→    setSaving(true);
   127→    try {
   128→      const res = await fetch("/api/designs", {
   129→        method: "POST",
   130→        headers: { "Content-Type": "application/json" },
   131→        body: JSON.stringify({ ...design, ...pack }),
   132→      });
   133→      const data = await res.json();
   134→      if (!res.ok) throw new Error(data.error);
   135→      toast({ title: "Design saved", description: `${design.name} is stored in the database.` });
   136→      loadSaved();
   137→    } catch (e) {
   138→      toast({ title: "Save failed", description: String(e instanceof Error ? e.message : e), variant: "destructive" });
   139→    } finally {
   140→      setSaving(false);
   141→    }
   142→  };
   143→
   144→  const load = (d: SavedDesign) => {
   145→    const n = (v: number | null | undefined, dflt = 0) => (v === null || v === undefined ? dflt : v);
   146→    const s = (v: string | null) => v ?? "";
   147→    const loaded: Partial<CellDesign> = {
   148→      name: d.name, chemistry: d.chemistry as CellDesign["chemistry"],
   149→      brand: s(d.brand), model: s(d.model),
   150→      formFactor: s(d.formFactor), cathode: s(d.cathode), anode: s(d.anode),
   151→      separator: s(d.separator), electrolyte: s(d.electrolyte), application: s(d.application),
   152→      sourceRef: s(d.sourceRef),
   153→      capacityAh: d.capacityAh, nominalV: n(d.nominalV), vmax: d.vmax, vmin: d.vmin, r0: d.r0,
   154→      r1: n(d.r1), c1: n(d.c1), r2: n(d.r2), c2: n(d.c2),
   155→      maxDischargeC: n(d.maxDischargeC), maxPulseDischargeC: n(d.maxPulseDischargeC),
   156→      maxChargeC: n(d.maxChargeC), stdChargeC: n(d.stdChargeC),
   157→      diameterMm: n(d.diameterMm), lengthMm: n(d.lengthMm), widthMm: n(d.widthMm),
   158→      heightMm: n(d.heightMm), thicknessMm: n(d.thicknessMm), massKg: d.massKg,
   159→      cpJkgK: n(d.cpJkgK), areaM2: n(d.areaM2), entropicVperK: n(d.entropicVperK, 0.0001),
   160→      cycleLifeCycles: n(d.cycleLifeCycles), cycleLifeNote: s(d.cycleLifeNote),
   161→      chargeTempMinC: n(d.chargeTempMinC), chargeTempMaxC: n(d.chargeTempMaxC),
   162→      dischargeTempMinC: n(d.dischargeTempMinC), dischargeTempMaxC: n(d.dischargeTempMaxC),
   163→      selfDischargePctMonth: n(d.selfDischargePctMonth),
   164→    };
   165→    updateDesign(loaded);
   166→    useLabStore.getState().setPack({ series: d.seriesCount, parallel: d.parallelCount });
   167→    toast({ title: "Design loaded", description: `${d.name} — all saved values restored.` });
   168→  };
   169→
   170→  const remove = async (id: string) => {
   171→    try {
   172→      await fetch(`/api/designs/${id}`, { method: "DELETE" });
   173→      loadSaved();
   174→      toast({ title: "Design deleted" });
   175→    } catch {
   176→      toast({ title: "Delete failed", variant: "destructive" });
   177→    }
   178→  };
   179→
   180→  const askAi = (prompt: string) => sendCopilotMessage(prompt);
   181→
   182→  const pickCell = (id: string) => {

   184→    const cell = getLibraryCell(id);
   185→    if (!cell) return;

   187→    toast({
   188→      title: "Datasheet loaded",
   189→      description: `${cell.brand} ${cell.model} — every value below is now editable.`,
   190→    });
   191→  };
   192→
   193→  return (
   194→    <div className="space-y-6">
   195→      <SectionTitle
   196→        icon={<BatteryCharging className="h-5 w-5" />}

   201→      {/* Starting point: blank or from datasheet */}

   204→          <div className="flex-1">
   205→            <div className="text-sm font-medium flex items-center gap-2">
   206→              <BookOpen className="h-4 w-4 text-primary" /> Starting point
   207→            </div>
   208→            <p className="text-xs text-muted-foreground mt-0.5">
   209→              Pre-fill every field from {LIBRARY_CELLS.length} real datasheets, or start blank — loaded values stay editable.
   210→            </p>
   211→          </div>
   212→          <div className="flex flex-col sm:flex-row gap-2 lg:w-[520px]">
   213→            <Select value={pickId} onValueChange={pickCell}>
   214→              <SelectTrigger className="h-9 flex-1" aria-label="Pick a datasheet as starting point">
   215→                <SelectValue placeholder="Load from datasheet…" />
   216→              </SelectTrigger>
   217→              <SelectContent>
   218→                {LIBRARY_CATEGORIES.map((cat) => {
   219→                  const cells = LIBRARY_CELLS.filter((c) => c.category === cat);
   220→                  if (cells.length === 0) return null;
   221→                  return (
   222→                    <SelectGroup key={cat}>
   223→                      <SelectLabel>{cat}</SelectLabel>
   224→                      {cells.map((c: LibraryCell) => (
   225→                        <SelectItem key={c.id} value={c.id}>
   226→                          {c.brand} — {c.model} ({c.capacityAh} Ah / {c.nominalV} V)
   227→                        </SelectItem>
   228→                      ))}
   229→                    </SelectGroup>
   230→                  );
   231→                })}
   232→              </SelectContent>
   233→            </Select>
   234→            <Button variant="outline" onClick={() => { resetDesign(); setPickId(""); toast({ title: "Design reset to blank sheet" }); }}>
   235→              <RotateCcw className="h-3.5 w-3.5 mr-1" /> Blank sheet
   236→            </Button>
   237→          </div>
   238→        </CardContent>
   239→      </Card>
   240→
   241→      {/* Completeness checklist */}
   242→      <Card className={complete ? "border-primary/40 bg-primary/[0.04]" : ""}>
   243→        <CardContent className="p-4">
   244→          <div className="flex items-center justify-between gap-3">
   245→            <div className="text-sm font-medium flex items-center gap-2">
   246→              {complete ? (
   247→                <><CheckCircle2 className="h-4 w-4 text-primary" /> Core definition complete — next: Connection, then Simulation</>
   248→              ) : (
   249→                <><CircleDashed className="h-4 w-4 text-amber-600" /> Design checklist — {issues.length} to go</>
   250→              )}
   251→            </div>
   252→            {design.sourceRef ? (
   253→              <Badge variant="outline" className="text-[10px] max-w-[280px] truncate">{design.sourceRef}</Badge>
   254→            ) : null}
   255→          </div>
   256→          {!complete && (
   257→            <ul className="grid sm:grid-cols-2 gap-x-4 gap-y-1 mt-2">
   258→              {issues.map((i) => (
   259→                <li key={i} className="text-xs text-muted-foreground flex items-center gap-1.5">
   260→                  <span className="h-1.5 w-1.5 rounded-full bg-amber-500 shrink-0" /> {i}
   261→                </li>
   262→              ))}
   263→            </ul>
   264→          )}
   265→        </CardContent>
   266→      </Card>
   267→      <div className="flex flex-wrap gap-2">
   268→        <button
   269→          onClick={() => askAi("I'm starting from a blank sheet. Guide me step by step — first ask me what I'm building.")}
   270→          disabled={copilotBusy}
   271→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   272→        >
   273→          <Sparkles className="h-3.5 w-3.5" /> Guide me from zero
   274→        </button>
   275→        <button
   276→          onClick={() => askAi("Recommend a real cell from the Cell Library for my application and explain why, with the apply block so I can load it.")}
   277→          disabled={copilotBusy}
   278→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   279→        >
   280→          <Sparkles className="h-3.5 w-3.5" /> Recommend a real cell
   281→        </button>
   282→        <button
   283→          onClick={() => askAi("Suggest a complete cell parameter set for my application (I'll describe it next), including the apply block so I can load it with one click.")}
   284→          disabled={copilotBusy}
   285→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   286→        >
   287→          <Sparkles className="h-3.5 w-3.5" /> Suggest values for my application
   288→        </button>
   289→        <button
   290→          onClick={() => askAi("Review my current cell parameters and warn me about anything unsafe or unrealistic.")}
   291→          disabled={copilotBusy}
   292→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   293→        >
   294→          <Sparkles className="h-3.5 w-3.5" /> Review my numbers
   295→        </button>
   296→      </div>
   297→
   298→      {/* ===== Input sections ===== */}
   299→      <div className="grid lg:grid-cols-2 gap-4">
   300→        {/* Identity & materials */}
   301→        <Card>
   302→          <CardHeader className="pb-2">
   303→            <CardTitle className="text-sm flex items-center gap-2"><BatteryCharging className="h-4 w-4 text-primary" /> Identity &amp; materials</CardTitle>
   304→            <CardDescription>What the battery is and what it&apos;s made of</CardDescription>
   305→          </CardHeader>
   306→          <CardContent className="grid grid-cols-2 gap-3">
   307→            <div className="col-span-2">
   308→              <Label htmlFor="dname">Design name</Label>
   309→              <Input id="dname" className="mt-1" placeholder="e.g. My 21700 traction cell" value={design.name} onChange={(e) => updateDesign({ name: e.target.value })} />
   310→            </div>
   311→            <div>
   312→              <TextField label="Brand (based on)" value={design.brand} placeholder="e.g. Samsung SDI" onChange={(v) => updateDesign({ brand: v })} />
   313→            </div>
   314→            <div>
   315→              <TextField label="Model (based on)" value={design.model} placeholder="e.g. INR21700-40T" onChange={(v) => updateDesign({ model: v })} />
   316→            </div>
   317→            <div className="col-span-2">
   318→              <Label className="text-xs text-muted-foreground">Chemistry (OCV curve family)</Label>
   319→              <Select value={design.chemistry} onValueChange={(v) => updateDesign({ chemistry: v as Chemistry })}>
   320→                <SelectTrigger className="mt-1 h-9"><SelectValue /></SelectTrigger>
   321→                <SelectContent>
   322→                  {CHEM_OPTIONS.map((c) => (
   323→                    <SelectItem key={c.id} value={c.id}>
   324→                      <span className="font-medium">{c.label}</span>
   325→                      <span className="text-muted-foreground text-xs"> — {c.hint}</span>
   326→                    </SelectItem>
   327→                  ))}
   328→                </SelectContent>
   329→              </Select>
   330→            </div>
   331→            <div className="col-span-2">
   332→              <Label className="text-xs text-muted-foreground">Form factor</Label>
   333→              <Select value={design.formFactor} onValueChange={(v) => updateDesign({ formFactor: v })}>
   334→                <SelectTrigger className="mt-1 h-9"><SelectValue placeholder="Select or leave custom" /></SelectTrigger>
   335→                <SelectContent>
   336→                  {FORM_FACTORS.map((f) => (
   337→                    <SelectItem key={f} value={f}>{f}</SelectItem>
   338→                  ))}
   339→                </SelectContent>
   340→              </Select>
   341→            </div>
   342→            <TextField label="Cathode (+) material" value={design.cathode} placeholder="e.g. NMC 811, LiFePO₄" onChange={(v) => updateDesign({ cathode: v })} />
   343→            <TextField label="Anode (−) material" value={design.anode} placeholder="e.g. Graphite + SiOx" onChange={(v) => updateDesign({ anode: v })} />
   344→            <TextField label="Separator" value={design.separator} placeholder="e.g. PE/PP microporous" onChange={(v) => updateDesign({ separator: v })} />
   345→            <TextField label="Electrolyte" value={design.electrolyte} placeholder="e.g. LiPF₆ carbonate" onChange={(v) => updateDesign({ electrolyte: v })} />
   346→            <div className="col-span-2">

   348→            </div>
   349→          </CardContent>
   350→        </Card>
   351→
   352→        {/* Geometry & mass */}
   353→        <Card>
   354→          <CardHeader className="pb-2">
   355→            <CardTitle className="text-sm flex items-center gap-2"><Ruler className="h-4 w-4 text-primary" /> Format, geometry &amp; mass</CardTitle>
   356→            <CardDescription>Physical envelope — also used to estimate surface area for thermal testing</CardDescription>
   357→          </CardHeader>
   358→          <CardContent className="grid grid-cols-2 gap-3">
   359→            <NumField label="Diameter Ø (mm)" value={design.diameterMm} step={0.05} min={0} onChange={(v) => updateDesign({ diameterMm: v })} hint="cylindrical cells" />

   362→            <NumField label="Height (mm)" value={design.heightMm} step={0.1} min={0} onChange={(v) => updateDesign({ heightMm: v })} hint="prismatic" />
   363→            <NumField label="Thickness (mm)" value={design.thicknessMm} step={0.1} min={0} onChange={(v) => updateDesign({ thicknessMm: v })} hint="pouch" />

   365→            <div className="col-span-2 flex items-center justify-between gap-2 rounded-lg bg-muted/50 px-3 py-2">
   366→              <p className="text-xs text-muted-foreground">
   367→                Estimated from dimensions: {estimateAreaM2(design) > 0 ? <span className="font-medium text-foreground">{(estimateAreaM2(design) * 10000).toFixed(0)} cm²</span> : "add dimensions"}

   370→                size="sm" variant="outline"
   371→                disabled={estimateAreaM2(design) <= 0}
   372→                onClick={() => {

   376→              >
   377→                <Ruler className="h-3.5 w-3.5 mr-1" /> Apply estimate
   378→              </Button>
   379→            </div>
   380→          </CardContent>
   381→        </Card>
   382→
   383→        {/* Electrical ratings */}
   384→        <Card>
   385→          <CardHeader className="pb-2">
   386→            <CardTitle className="text-sm flex items-center gap-2"><Gauge className="h-4 w-4 text-primary" /> Electrical ratings</CardTitle>
   387→            <CardDescription>Everything the datasheet front page lists</CardDescription>
   388→          </CardHeader>
   389→          <CardContent className="grid grid-cols-2 gap-3">
   390→            <NumField label="Capacity (Ah)" value={design.capacityAh} step={0.1} min={0} onChange={(v) => updateDesign({ capacityAh: v })} />

   392→            <NumField label="V max — charge cutoff (V)" value={design.vmax} step={0.05} min={0} onChange={(v) => updateDesign({ vmax: v })} />

   498→// ------------------------------------------------------------
   499→// Real-world condition preview — computed live from the
   500→// datasheet values the user entered (OCV curve + R0 + mass).
   501→// ------------------------------------------------------------
   502→
   503→function RealWorldPreview() {
   504→  const design = useLabStore((s) => s.design);
   505→  const ready = design.capacityAh > 0 && design.r0 > 0 && design.vmin > 0 && design.vmax > design.vmin;
   506→  const maxC = Math.min(design.maxDischargeC > 0 ? design.maxDischargeC : 3, 12);
   507→
   508→  // Loaded voltage vs SOC at several C-rates
   509→  const sagData = useMemo(() => {
   510→    if (!ready) return [];
   511→    const rates = [0.25, 0.5, 1, 2, maxC].filter((r, i, arr) => arr.indexOf(r) === i).sort((a, b) => a - b);
   512→    const pts: { soc: number; [k: string]: number }[] = [];
   513→    for (let soc = 100; soc >= 0; soc -= 2) {
   514→      const row: { soc: number; [k: string]: number } = { soc };
   515→      for (const r of rates) row[`${r}C`] = +loadedVoltageAtSoc(design, soc / 100, r).toFixed(3);
   516→      pts.push(row);
   517→    }
   518→    return pts;
   519→  }, [design, ready, maxC]);
   520→
   521→  // Effective capacity vs C-rate
   522→  const effData = useMemo(() => {
   523→    if (!ready) return [];
   524→    const pts: { cRate: number; usable: number | null }[] = [];
   525→    const step = maxC <= 2 ? 0.1 : 0.25;
   526→    for (let c = 0.1; c <= maxC + 1e-9; c += step) {
   527→      pts.push({ cRate: +c.toFixed(2), usable: effectiveCapacityPct(design, +c.toFixed(2)) });
   528→    }
   529→    pts.push({ cRate: +maxC.toFixed(2), usable: effectiveCapacityPct(design, maxC) });
   530→    return pts;
   531→  }, [design, ready, maxC]);
   532→
   533→  // voltage sag at 1C: pure ohmic drop I·R0 (1C current = capacity in amps)
   534→  const sag1C = ready ? design.capacityAh * design.r0 * 1000 : 0;
   535→  const heatMax = jouleHeatW(design, maxC);
   536→  const dT30 = tempRiseAdiabatic(design, maxC, 1800);
   537→  const runtime1C = ready ? +(60 * (effectiveCapacityPct(design, 1) ?? 0) / 100).toFixed(0) : null;
   538→  const usableAtMax = ready ? effectiveCapacityPct(design, maxC) : null;
   539→
   540→  if (!ready) {
   541→    return (
   542→      <Card>
   543→        <CardHeader className="pb-2">
   544→          <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-primary" /> Real-World Preview</CardTitle>
   545→          <CardDescription>Enter capacity, R₀ and the voltage window to activate the preview.</CardDescription>
   546→        </CardHeader>
   547→      </Card>
   548→    );
   549→  }
   550→
   551→  return (
   552→    <Card>
   553→      <CardHeader className="pb-2">
   554→        <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-primary" /> Real-World Preview</CardTitle>
   555→        <CardDescription>
   556→          Live from your values — ohmic estimate; full electro-thermal runs in Simulation.
   557→        </CardDescription>
   558→      </CardHeader>
   559→      <CardContent className="space-y-4">
   560→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   561→          <StatCard
   562→            label="Sag @1C (ohmic)"
   563→            value={sag1C.toFixed(0)} unit="mV"
   564→            hint="V drop below OCV from I·R₀"
   565→          />
   566→          <StatCard
   567→            label={`Heat @ ${maxC}C`}
   568→            value={heatMax ?? "—"} unit={heatMax != null ? "W" : undefined}
   569→            hint="I²·R₀ joule heating"
   570→            tone={heatMax != null && heatMax > 5 ? "warn" : "default"}
   571→          />
   572→          <StatCard
   573→            label={`ΔT @ ${maxC}C, 30 min`}
   574→            value={dT30 ?? "—"} unit={dT30 != null ? "°C" : undefined}
   575→            hint="adiabatic estimate"
   576→            tone={dT30 != null && dT30 > 25 ? "fail" : dT30 != null && dT30 > 10 ? "warn" : "default"}
   577→          />
   578→          <StatCard
   579→            label="Runtime @ 1C → Vmin"
   580→            value={runtime1C != null && runtime1C > 0 ? runtime1C : "—"} unit={runtime1C != null && runtime1C > 0 ? "min" : undefined}
   581→            hint="full charge to cutoff"
   582→          />
   583→        </div>
   584→
   585→        <div className="grid lg:grid-cols-2 gap-4">
   586→          <ChartCard title="Loaded voltage vs SOC — voltage sag under real loads">
   587→            <ResponsiveContainer width="100%" height="100%">
   588→              <LineChart data={sagData} margin={{ top: 8, right: 16, bottom: 8, left: 0 }}>
   589→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
   590→                <XAxis dataKey="soc" tick={{ fontSize: 11 }} unit="%" />
   591→                <YAxis domain={["auto", "auto"]} tick={{ fontSize: 11 }} unit=" V" />
   592→                <Tooltip formatter={(v: number | string) => [`${v} V`, ""]} labelFormatter={(l) => `SOC ${l}%`} />
   593→                <Legend wrapperStyle={{ fontSize: 11 }} />
   594→                <ReferenceLine y={design.vmin} stroke={CHART_COLORS.red} strokeDasharray="4 4" label={{ value: "Vmin", fontSize: 10, fill: CHART_COLORS.red }} />
   595→                {[0.25, 0.5, 1, 2, maxC].filter((r, i, a) => a.indexOf(r) === i).map((r, i) => (
   596→                  <Line
   597→                    key={r}
   598→                    type="monotone" dataKey={`${r}C`} name={`${r}C (${(r * design.capacityAh).toFixed(1)} A)`}
   599→                    stroke={[CHART_COLORS.blue, CHART_COLORS.cyan, CHART_COLORS.slate, CHART_COLORS.amber, CHART_COLORS.red][i % 5]}
   600→                    strokeWidth={2} dot={false}
   601→                  />
   602→                ))}
   603→              </LineChart>
   604→            </ResponsiveContainer>
   605→          </ChartCard>
   606→
   607→          <ChartCard title="Usable capacity vs C-rate — real-world derating">
   608→            <ResponsiveContainer width="100%" height="100%">
   609→              <LineChart data={effData} margin={{ top: 8, right: 16, bottom: 8, left: 0 }}>
   610→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
   611→                <XAxis dataKey="cRate" tick={{ fontSize: 11 }} unit="C" />
   612→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} unit="%" />
   613→                <Tooltip formatter={(v: number | string) => [`${v} %`, "usable"]} labelFormatter={(l) => `${l}C`} />
   614→                <Legend wrapperStyle={{ fontSize: 11 }} />
   615→                <Line type="monotone" dataKey="usable" name="Usable capacity" stroke={CHART_COLORS.cyan} strokeWidth={2.5} dot={false} />
   616→              </LineChart>
   617→            </ResponsiveContainer>
   618→          </ChartCard>
   619→        </div>
   620→
   621→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   622→          <TrendingDown className="h-3.5 w-3.5 text-primary shrink-0" />
   623→          <span>≈<span className="font-semibold text-foreground">{usableAtMax ?? "—"}%</span> of rated capacity is usable at {maxC}C before the loaded voltage hits Vmin.</span>
   624→        </div>
   625→      </CardContent>
   626→    </Card>
   627→  );
   628→}

   634→function NumField({ label, value, step, min, onChange, hint }: { label: string; value: number; step: number; min?: number; onChange: (v: number) => void; hint?: string }) {
   635→  return (
   636→    <div>
   637→      <Label className="text-xs text-muted-foreground">{label}</Label>
   638→      <Input
   639→        className="mt-1 h-8"
   640→        type="number"
   641→        value={value || ""}
   642→        placeholder="0"
   643→        step={step}
   644→        min={min}
   645→        onChange={(e) => {
   646→          const v = parseFloat(e.target.value);
   647→          if (!isNaN(v)) onChange(v);
   648→          else if (e.target.value === "") onChange(0);
   649→        }}
   650→      />
   651→      {hint && <p className="text-[10px] text-muted-foreground mt-0.5">{hint}</p>}
   652→    </div>
   653→  );
   654→}

   656→function TextField({ label, value, placeholder, onChange }: { label: string; value: string; placeholder?: string; onChange: (v: string) => void }) {
   657→  return (

   659→      <Label className="text-xs text-muted-foreground">{label}</Label>
   660→      <Input
   661→        className="mt-1 h-8"
   662→        value={value}
   663→        placeholder={placeholder}
   664→        onChange={(e) => onChange(e.target.value)}
   665→      />
   666→    </div>
   667→  );
