    94→export function DesignTab() {
    95→  const { design, updateDesign, resetDesign, pack, sendCopilotMessage, copilotBusy } = useLabStore();
    96→  const { toast } = useToast();
    97→  const [saved, setSaved] = useState<SavedDesign[]>([]);
    98→  const [saving, setSaving] = useState(false);
    99→  const [pickId, setPickId] = useState<string>("");
   100→
   101→  const loadSaved = async () => {
   102→    try {
   103→      const res = await fetch("/api/designs");
   104→      const data = await res.json();
   105→      if (res.ok) setSaved(data.designs);
   106→    } catch {
   107→      /* silent */
   108→    }
   109→  };
   110→
   111→  useEffect(() => {
   112→    loadSaved();
   113→  }, []);
   114→
   115→  const issues = designIssues(design);
   116→  const complete = issues.length === 0;
   117→  const pm = packMath(design, pack);
   118→  const cellWh = design.capacityAh * (design.nominalV > 0 ? design.nominalV : (design.vmax + design.vmin) / 2 + 0.05);
   119→  const whKg = design.massKg > 0 ? cellWh / design.massKg : 0;
   120→
   121→  const save = async () => {
   122→    if (!complete) {
   123→      toast({ title: "Design incomplete", description: "Fill every required field before saving.", variant: "destructive" });
   124→      return;
   125→    }
   126→    setSaving(true);
   127→    try {
   128→      const res = await fetch("/api/designs", {
   129→        method: "POST",
   130→        headers: { "Content-Type": "application/json" },
   131→        body: JSON.stringify({ ...design, ...pack }),
   132→      });
   133→      const data = await res.json();
   134→      if (!res.ok) throw new Error(data.error);
   135→      toast({ title: "Design saved", description: `${design.name} is stored in the database.` });
   136→      loadSaved();
   137→    } catch (e) {
   138→      toast({ title: "Save failed", description: String(e instanceof Error ? e.message : e), variant: "destructive" });
   139→    } finally {
   140→      setSaving(false);
   141→    }
   142→  };
   143→
   144→  const load = (d: SavedDesign) => {
   145→    const n = (v: number | null | undefined, dflt = 0) => (v === null || v === undefined ? dflt : v);
   146→    const s = (v: string | null) => v ?? "";
   147→    const loaded: Partial<CellDesign> = {
   148→      name: d.name, chemistry: d.chemistry as CellDesign["chemistry"],
   149→      brand: s(d.brand), model: s(d.model),
   150→      formFactor: s(d.formFactor), cathode: s(d.cathode), anode: s(d.anode),
   151→      separator: s(d.separator), electrolyte: s(d.electrolyte), application: s(d.application),
   152→      sourceRef: s(d.sourceRef),
   153→      capacityAh: d.capacityAh, nominalV: n(d.nominalV), vmax: d.vmax, vmin: d.vmin, r0: d.r0,
   154→      r1: n(d.r1), c1: n(d.c1), r2: n(d.r2), c2: n(d.c2),
   155→      maxDischargeC: n(d.maxDischargeC), maxPulseDischargeC: n(d.maxPulseDischargeC),
   156→      maxChargeC: n(d.maxChargeC), stdChargeC: n(d.stdChargeC),
   157→      diameterMm: n(d.diameterMm), lengthMm: n(d.lengthMm), widthMm: n(d.widthMm),
   158→      heightMm: n(d.heightMm), thicknessMm: n(d.thicknessMm), massKg: d.massKg,
   159→      cpJkgK: n(d.cpJkgK), areaM2: n(d.areaM2), entropicVperK: n(d.entropicVperK, 0.0001),
   160→      cycleLifeCycles: n(d.cycleLifeCycles), cycleLifeNote: s(d.cycleLifeNote),
   161→      chargeTempMinC: n(d.chargeTempMinC), chargeTempMaxC: n(d.chargeTempMaxC),
   162→      dischargeTempMinC: n(d.dischargeTempMinC), dischargeTempMaxC: n(d.dischargeTempMaxC),
   163→      selfDischargePctMonth: n(d.selfDischargePctMonth),
   164→    };
   165→    updateDesign(loaded);
   166→    useLabStore.getState().setPack({ series: d.seriesCount, parallel: d.parallelCount });
   167→    toast({ title: "Design loaded", description: `${d.name} — all saved values restored.` });
   168→  };
   169→
   170→  const remove = async (id: string) => {
   171→    try {
   172→      await fetch(`/api/designs/${id}`, { method: "DELETE" });
   173→      loadSaved();
   174→      toast({ title: "Design deleted" });
   175→    } catch {
   176→      toast({ title: "Delete failed", variant: "destructive" });
   177→    }
   178→  };
   179→
   180→  const askAi = (prompt: string) => sendCopilotMessage(prompt);
   181→
   182→  const pickCell = (id: string) => {
   183→    setPickId(id);
   184→    const cell = getLibraryCell(id);
   185→    if (!cell) return;
   186→    useLabStore.getState().loadLibraryCell(cell);
   187→    toast({
   188→      title: "Datasheet loaded",
   189→      description: `${cell.brand} ${cell.model} — every value below is now editable.`,
   190→    });
   191→  };
   192→
   193→  return (
   194→    <div className="space-y-6">
   195→      <SectionTitle
   196→        icon={<BatteryCharging className="h-5 w-5" />}
   197→        title="1 · Battery Design — every input, your battery"
   198→        desc="This page carries ALL the inputs required to make a battery: identity, chemistry & materials, format, geometry, electrical ratings, temperature limits, durability and the thermal model. Start from a real datasheet in the Cell Library or from a blank sheet — the values you enter feed the pack connection, the model, the simulations and the thermal tests. VoltAI (bottom-right) guides you in real time."
   199→      />
   200→
   201→      {/* Starting point: blank or from datasheet */}
   202→      <Card className="border-emerald-500/40 bg-emerald-500/5">
   203→        <CardContent className="p-4 flex flex-col lg:flex-row lg:items-center gap-3">
   204→          <div className="flex-1">
   205→            <div className="text-sm font-medium flex items-center gap-2">
   206→              <BookOpen className="h-4 w-4 text-emerald-600" /> Starting point
   207→            </div>
   208→            <p className="text-xs text-muted-foreground mt-0.5 leading-relaxed">
   209→              Begin from a blank sheet, or pre-fill every field from one of {LIBRARY_CELLS.length} real datasheets
   210→              (Samsung, Panasonic, LG, Molicel, EVE, CATL, A123, Toshiba, Trojan, Yuasa…). Loaded values remain
   211→              fully editable — the design stays yours.
   212→            </p>
   213→          </div>
   214→          <div className="flex flex-col sm:flex-row gap-2 lg:w-[520px]">
   215→            <Select value={pickId} onValueChange={pickCell}>
   216→              <SelectTrigger className="h-9 flex-1" aria-label="Pick a datasheet as starting point">
   217→                <SelectValue placeholder="Load from datasheet…" />
   218→              </SelectTrigger>
   219→              <SelectContent>
   220→                {LIBRARY_CATEGORIES.map((cat) => {
   221→                  const cells = LIBRARY_CELLS.filter((c) => c.category === cat);
   222→                  if (cells.length === 0) return null;
   223→                  return (
   224→                    <SelectGroup key={cat}>
   225→                      <SelectLabel>{cat}</SelectLabel>
   226→                      {cells.map((c: LibraryCell) => (
   227→                        <SelectItem key={c.id} value={c.id}>
   228→                          {c.brand} — {c.model} ({c.capacityAh} Ah / {c.nominalV} V)
   229→                        </SelectItem>
   230→                      ))}
   231→                    </SelectGroup>
   232→                  );
   233→                })}
   234→              </SelectContent>
   235→            </Select>
   236→            <Button variant="outline" onClick={() => { resetDesign(); setPickId(""); toast({ title: "Design reset to blank sheet" }); }}>
   237→              <RotateCcw className="h-3.5 w-3.5 mr-1" /> Blank sheet
   238→            </Button>
   239→          </div>
   240→        </CardContent>
   241→      </Card>
   242→
   243→      {/* Completeness checklist */}
   244→      <Card className={complete ? "border-emerald-500/50 bg-emerald-500/5" : ""}>
   245→        <CardContent className="p-4">
   246→          <div className="flex items-center justify-between gap-3 mb-2">
   247→            <div className="text-sm font-medium flex items-center gap-2">
   248→              {complete ? (
   249→                <><CheckCircle2 className="h-4 w-4 text-emerald-600" /> Core definition complete — head to Connection, or review the Real-World Preview below</>
   250→              ) : (
   251→                <><CircleDashed className="h-4 w-4 text-amber-600" /> Your design checklist ({issues.length} to go)</>
   252→              )}
   253→            </div>
   254→            {design.sourceRef ? (
   255→              <Badge variant="outline" className="text-[10px] max-w-[280px] truncate">{design.sourceRef}</Badge>
   256→            ) : null}
   257→          </div>
   258→          {complete ? (
   259→            <p className="text-xs text-muted-foreground">
   260→              Next: check the <span className="font-medium text-foreground">Real-World Preview</span> at the bottom of this page, then set your
   261→              series × parallel arrangement in <span className="font-medium text-foreground">Connection</span>.
   262→            </p>
   263→          ) : (
   264→            <ul className="grid sm:grid-cols-2 gap-x-4 gap-y-1">
   265→              {issues.map((i) => (
   266→                <li key={i} className="text-xs text-muted-foreground flex items-center gap-1.5">
   267→                  <span className="h-1.5 w-1.5 rounded-full bg-amber-500 shrink-0" /> {i}
   268→                </li>
   269→              ))}
   270→            </ul>
   271→          )}
   272→        </CardContent>
   273→      </Card>
   274→
   275→      {/* AI assist chips */}
   276→      <div className="flex flex-wrap gap-2">
   277→        <button
   278→          onClick={() => askAi("I'm starting from a blank sheet. Guide me step by step — first ask me what I'm building.")}
   279→          disabled={copilotBusy}
   280→          className="inline-flex items-center gap-1.5 rounded-full border border-emerald-500/50 bg-emerald-500/5 px-3 py-1.5 text-xs font-medium text-emerald-700 hover:bg-emerald-500/10 transition-colors disabled:opacity-50"
   281→        >
   282→          <Sparkles className="h-3.5 w-3.5" /> Guide me from zero
   283→        </button>
   284→        <button
   285→          onClick={() => askAi("Recommend a real cell from the Cell Library for my application and explain why, with the apply block so I can load it.")}
   286→          disabled={copilotBusy}
   287→          className="inline-flex items-center gap-1.5 rounded-full border border-emerald-500/50 bg-emerald-500/5 px-3 py-1.5 text-xs font-medium text-emerald-700 hover:bg-emerald-500/10 transition-colors disabled:opacity-50"
   288→        >
   289→          <Sparkles className="h-3.5 w-3.5" /> Recommend a real cell
   290→        </button>
   291→        <button
   292→          onClick={() => askAi("Suggest a complete cell parameter set for my application (I'll describe it next), including the apply block so I can load it with one click.")}
   293→          disabled={copilotBusy}
   294→          className="inline-flex items-center gap-1.5 rounded-full border border-emerald-500/50 bg-emerald-500/5 px-3 py-1.5 text-xs font-medium text-emerald-700 hover:bg-emerald-500/10 transition-colors disabled:opacity-50"
   295→        >
   296→          <Sparkles className="h-3.5 w-3.5" /> Suggest values for my application
   297→        </button>
   298→        <button
   299→          onClick={() => askAi("Review my current cell parameters and warn me about anything unsafe or unrealistic.")}
   300→          disabled={copilotBusy}
   301→          className="inline-flex items-center gap-1.5 rounded-full border border-emerald-500/50 bg-emerald-500/5 px-3 py-1.5 text-xs font-medium text-emerald-700 hover:bg-emerald-500/10 transition-colors disabled:opacity-50"
   302→        >
   303→          <Sparkles className="h-3.5 w-3.5" /> Review my numbers
   304→        </button>
   305→      </div>
   306→
   307→      {/* ===== Input sections ===== */}
   308→      <div className="grid lg:grid-cols-2 gap-4">
   309→        {/* Identity & materials */}
   310→        <Card>
   311→          <CardHeader className="pb-2">
   312→            <CardTitle className="text-sm flex items-center gap-2"><BatteryCharging className="h-4 w-4 text-emerald-600" /> Identity &amp; materials</CardTitle>
   313→            <CardDescription>What the battery is and what it&apos;s made of</CardDescription>
   314→          </CardHeader>
   315→          <CardContent className="grid grid-cols-2 gap-3">
   316→            <div className="col-span-2">
   317→              <Label htmlFor="dname">Design name</Label>
   318→              <Input id="dname" className="mt-1" placeholder="e.g. My 21700 traction cell" value={design.name} onChange={(e) => updateDesign({ name: e.target.value })} />
   319→            </div>
   320→            <div>
   321→              <TextField label="Brand (based on)" value={design.brand} placeholder="e.g. Samsung SDI" onChange={(v) => updateDesign({ brand: v })} />
   322→            </div>
   323→            <div>
   324→              <TextField label="Model (based on)" value={design.model} placeholder="e.g. INR21700-40T" onChange={(v) => updateDesign({ model: v })} />
   325→            </div>
   326→            <div className="col-span-2">
   327→              <Label className="text-xs text-muted-foreground">Chemistry (OCV curve family)</Label>
   328→              <Select value={design.chemistry} onValueChange={(v) => updateDesign({ chemistry: v as Chemistry })}>
   329→                <SelectTrigger className="mt-1 h-9"><SelectValue /></SelectTrigger>
   330→                <SelectContent>
   331→                  {CHEM_OPTIONS.map((c) => (
   332→                    <SelectItem key={c.id} value={c.id}>
   333→                      <span className="font-medium">{c.label}</span>
   334→                      <span className="text-muted-foreground text-xs"> — {c.hint}</span>
   335→                    </SelectItem>
   336→                  ))}
   337→                </SelectContent>
   338→              </Select>
   339→            </div>
   340→            <div className="col-span-2">

   344→                <SelectContent>
   345→                  {FORM_FACTORS.map((f) => (
   346→                    <SelectItem key={f} value={f}>{f}</SelectItem>
   347→                  ))}
   348→                </SelectContent>
   349→              </Select>
   350→            </div>
   351→            <TextField label="Cathode (+) material" value={design.cathode} placeholder="e.g. NMC 811, LiFePO₄" onChange={(v) => updateDesign({ cathode: v })} />
   352→            <TextField label="Anode (−) material" value={design.anode} placeholder="e.g. Graphite + SiOx" onChange={(v) => updateDesign({ anode: v })} />

   357→            </div>
   358→          </CardContent>
   359→        </Card>

   361→        {/* Geometry & mass */}
   362→        <Card>
   363→          <CardHeader className="pb-2">
   364→            <CardTitle className="text-sm flex items-center gap-2"><Ruler className="h-4 w-4 text-emerald-600" /> Format, geometry &amp; mass</CardTitle>
   365→            <CardDescription>Physical envelope — also used to estimate surface area for thermal testing</CardDescription>
   366→          </CardHeader>
   367→          <CardContent className="grid grid-cols-2 gap-3">
   368→            <NumField label="Diameter Ø (mm)" value={design.diameterMm} step={0.05} min={0} onChange={(v) => updateDesign({ diameterMm: v })} hint="cylindrical cells" />
   369→            <NumField label="Length / height (mm)" value={design.lengthMm} step={0.1} min={0} onChange={(v) => updateDesign({ lengthMm: v })} hint="cylinder length" />
   370→            <NumField label="Width (mm)" value={design.widthMm} step={0.1} min={0} onChange={(v) => updateDesign({ widthMm: v })} hint="prismatic" />
   371→            <NumField label="Height (mm)" value={design.heightMm} step={0.1} min={0} onChange={(v) => updateDesign({ heightMm: v })} hint="prismatic" />
   372→            <NumField label="Thickness (mm)" value={design.thicknessMm} step={0.1} min={0} onChange={(v) => updateDesign({ thicknessMm: v })} hint="pouch" />

   374→            <div className="col-span-2 flex items-center justify-between gap-2 rounded-lg bg-muted/50 px-3 py-2">
   375→              <p className="text-xs text-muted-foreground">
   376→                Surface area drives the thermal model — estimate it from the dimensions above.

   386→              >
   387→                <Ruler className="h-3.5 w-3.5 mr-1" /> Apply estimate
   388→              </Button>
   389→            </div>
   390→          </CardContent>
   391→        </Card>
   392→
   393→        {/* Electrical ratings */}
   394→        <Card>
   395→          <CardHeader className="pb-2">

   398→          </CardHeader>
   399→          <CardContent className="grid grid-cols-2 gap-3">

   516→function RealWorldPreview() {
   517→  const design = useLabStore((s) => s.design);
   518→  const ready = design.capacityAh > 0 && design.r0 > 0 && design.vmin > 0 && design.vmax > design.vmin;
   519→  const maxC = Math.min(design.maxDischargeC > 0 ? design.maxDischargeC : 3, 12);
   520→
   521→  // Loaded voltage vs SOC at several C-rates
   522→  const sagData = useMemo(() => {
   523→    if (!ready) return [];
   524→    const rates = [0.25, 0.5, 1, 2, maxC].filter((r, i, arr) => arr.indexOf(r) === i).sort((a, b) => a - b);
   525→    const pts: { soc: number; [k: string]: number }[] = [];
   526→    for (let soc = 100; soc >= 0; soc -= 2) {
   527→      const row: { soc: number; [k: string]: number } = { soc };
   528→      for (const r of rates) row[`${r}C`] = +loadedVoltageAtSoc(design, soc / 100, r).toFixed(3);
   529→      pts.push(row);
   530→    }
   531→    return pts;
   532→  }, [design, ready, maxC]);
   533→
   534→  // Effective capacity vs C-rate
   535→  const effData = useMemo(() => {
   536→    if (!ready) return [];
   537→    const pts: { cRate: number; usable: number | null }[] = [];
   538→    const step = maxC <= 2 ? 0.1 : 0.25;
   539→    for (let c = 0.1; c <= maxC + 1e-9; c += step) {
   540→      pts.push({ cRate: +c.toFixed(2), usable: effectiveCapacityPct(design, +c.toFixed(2)) });
   541→    }
   542→    pts.push({ cRate: +maxC.toFixed(2), usable: effectiveCapacityPct(design, maxC) });
   543→    return pts;
   544→  }, [design, ready, maxC]);
   545→
   546→  // voltage sag at 1C: pure ohmic drop I·R0 (1C current = capacity in amps)
   547→  const sag1C = ready ? design.capacityAh * design.r0 * 1000 : 0;
   548→  const heatMax = jouleHeatW(design, maxC);
   549→  const dT30 = tempRiseAdiabatic(design, maxC, 1800);
   550→  const runtime1C = ready ? +(60 * (effectiveCapacityPct(design, 1) ?? 0) / 100).toFixed(0) : null;
   551→  const usableAtMax = ready ? effectiveCapacityPct(design, maxC) : null;
   552→
   553→  if (!ready) {
   554→    return (
   555→      <Card>
   556→        <CardHeader className="pb-2">
   557→          <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-emerald-600" /> Real-World Preview — simulate datasheet behaviour while you design</CardTitle>
   558→          <CardDescription>Set capacity, R₀ and the voltage window above to activate the live preview.</CardDescription>
   559→        </CardHeader>
   560→      </Card>
   561→    );
   562→  }
   563→
   564→  return (
   565→    <Card>
   566→      <CardHeader className="pb-2">
   567→        <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-emerald-600" /> Real-World Preview — datasheet-condition behaviour</CardTitle>
   568→        <CardDescription>
   569→          Simulated live from YOUR values: the chemistry&apos;s OCV curve plus R₀ under load — like the
   570→          &quot;discharge characteristics / capacity vs rate&quot; curves on real datasheets. Ohmic-only estimate
   571→          (RC polarization adds more sag); full electro-thermal runs live in Simulation.
   572→        </CardDescription>
   573→      </CardHeader>
   574→      <CardContent className="space-y-4">
   575→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   576→          <StatCard
   577→            label="Sag @1C (ohmic)"
   578→            value={sag1C.toFixed(0)} unit="mV"
   579→            hint="V drop below OCV from I·R₀"
   580→          />
   581→          <StatCard
   582→            label={`Heat @ ${maxC}C`}
   583→            value={heatMax ?? "—"} unit={heatMax != null ? "W" : undefined}
   584→            hint="I²·R₀ joule heating"
   585→            tone={heatMax != null && heatMax > 5 ? "warn" : "default"}
   586→          />
   587→          <StatCard
   588→            label={`ΔT @ ${maxC}C, 30 min`}
   589→            value={dT30 ?? "—"} unit={dT30 != null ? "°C" : undefined}
   590→            hint="adiabatic estimate"
   591→            tone={dT30 != null && dT30 > 25 ? "fail" : dT30 != null && dT30 > 10 ? "warn" : "default"}
   592→          />
   593→          <StatCard
   594→            label="Runtime @ 1C → Vmin"
   595→            value={runtime1C != null && runtime1C > 0 ? runtime1C : "—"} unit={runtime1C != null && runtime1C > 0 ? "min" : undefined}
   596→            hint="full charge to cutoff"
   597→          />
   598→        </div>
   599→
   600→        <div className="grid lg:grid-cols-2 gap-4">
   601→          <ChartCard title="Loaded voltage vs SOC — voltage sag under real loads">
   602→            <ResponsiveContainer width="100%" height="100%">
   603→              <LineChart data={sagData} margin={{ top: 8, right: 16, bottom: 8, left: 0 }}>
   604→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
   605→                <XAxis dataKey="soc" tick={{ fontSize: 11 }} unit="%" />
   606→                <YAxis domain={["auto", "auto"]} tick={{ fontSize: 11 }} unit=" V" />
   607→                <Tooltip formatter={(v: number | string) => [`${v} V`, ""]} labelFormatter={(l) => `SOC ${l}%`} />
   608→                <Legend wrapperStyle={{ fontSize: 11 }} />
   609→                <ReferenceLine y={design.vmin} stroke={CHART_COLORS.red} strokeDasharray="4 4" label={{ value: "Vmin", fontSize: 10, fill: CHART_COLORS.red }} />
   610→                {[0.25, 0.5, 1, 2, maxC].filter((r, i, a) => a.indexOf(r) === i).map((r, i) => (
   611→                  <Line
   612→                    key={r}
   613→                    type="monotone" dataKey={`${r}C`} name={`${r}C (${(r * design.capacityAh).toFixed(1)} A)`}
   614→                    stroke={[CHART_COLORS.emerald, CHART_COLORS.teal, CHART_COLORS.slate, CHART_COLORS.amber, CHART_COLORS.red][i % 5]}
   615→                    strokeWidth={2} dot={false}
   616→                  />
   617→                ))}
   618→              </LineChart>
   619→            </ResponsiveContainer>
   620→          </ChartCard>
   621→
   622→          <ChartCard title="Usable capacity vs C-rate — real-world derating">
   623→            <ResponsiveContainer width="100%" height="100%">
   624→              <LineChart data={effData} margin={{ top: 8, right: 16, bottom: 8, left: 0 }}>
   625→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
   626→                <XAxis dataKey="cRate" tick={{ fontSize: 11 }} unit="C" />
   627→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} unit="%" />
   628→                <Tooltip formatter={(v: number | string) => [`${v} %`, "usable"]} labelFormatter={(l) => `${l}C`} />
   629→                <Legend wrapperStyle={{ fontSize: 11 }} />
   630→                <Line type="monotone" dataKey="usable" name="Usable capacity" stroke={CHART_COLORS.teal} strokeWidth={2.5} dot={false} />
   631→              </LineChart>
   632→            </ResponsiveContainer>
   633→          </ChartCard>
   634→        </div>
   635→
   636→        <div className="rounded-lg bg-muted/50 px-3 py-2 flex items-start gap-2 text-xs text-muted-foreground leading-relaxed">
   637→          <TrendingDown className="h-4 w-4 text-emerald-600 shrink-0 mt-0.5" />
   638→          <span>
   639→            At your {maxC}C maximum, ≈<span className="font-semibold text-foreground">{usableAtMax ?? "—"}%</span> of rated capacity is
   640→            reachable before the loaded voltage hits Vmin — the cell&apos;s real-world rating depends on the load, exactly as
   641→            datasheet capacity-vs-rate curves show. When ready, run full profiles (CC, CC-CV, HPPC, drive cycle or your own
   642→            custom sequence) in <span className="font-medium text-foreground">Simulation</span>, with heat and cooling in
   643→            <span className="font-medium text-foreground"> Thermal Test</span>.
   644→          </span>
   645→        </div>
   646→        <div className="flex items-center gap-2 text-[11px] text-muted-foreground">

   648→          estimates assume constant-current discharge from full.
   649→        </div>
   650→      </CardContent>
   651→    </Card>
   652→  );

   659→function NumField({ label, value, step, min, onChange, hint }: { label: string; value: number; step: number; min?: number; onChange: (v: number) => void; hint?: string }) {
   660→  return (
   661→    <div>
   662→      <Label className="text-xs text-muted-foreground">{label}</Label>
   663→      <Input
   664→        className="mt-1 h-8"
   665→        type="number"
   666→        value={value || ""}
   667→        placeholder="0"
   668→        step={step}
   669→        min={min}
   670→        onChange={(e) => {
   671→          const v = parseFloat(e.target.value);
   672→          if (!isNaN(v)) onChange(v);
   673→          else if (e.target.value === "") onChange(0);
   674→        }}
   675→      />
   676→      {hint && <p className="text-[10px] text-muted-foreground mt-0.5">{hint}</p>}
   677→    </div>
   678→  );
   679→}
