   503→   503→function RealWorldPreview() {
   504→   504→  const design = useLabStore((s) => s.design);
   505→   505→  const ready = design.capacityAh > 0 && design.r0 > 0 && design.vmin > 0 && design.vmax > design.vmin;
   506→   506→  const maxC = Math.min(design.maxDischargeC > 0 ? design.maxDischargeC : 3, 12);
   507→   507→
   508→   508→  // Loaded voltage vs SOC at several C-rates
   509→   509→  const sagData = useMemo(() => {
   510→   510→    if (!ready) return [];
   511→   511→    const rates = [0.25, 0.5, 1, 2, maxC].filter((r, i, arr) => arr.indexOf(r) === i).sort((a, b) => a - b);
   512→   512→    const pts: { soc: number; [k: string]: number }[] = [];
   513→   513→    for (let soc = 100; soc >= 0; soc -= 2) {
   514→   514→      const row: { soc: number; [k: string]: number } = { soc };
   515→   515→      for (const r of rates) row[`${r}C`] = +loadedVoltageAtSoc(design, soc / 100, r).toFixed(3);
   516→   516→      pts.push(row);
   517→   517→    }
   518→   518→    return pts;
   519→   519→  }, [design, ready, maxC]);
   520→   520→
   521→   521→  // Effective capacity vs C-rate
   522→   522→  const effData = useMemo(() => {
   523→   523→    if (!ready) return [];
   524→   524→    const pts: { cRate: number; usable: number | null }[] = [];
   525→   525→    const step = maxC <= 2 ? 0.1 : 0.25;
   526→   526→    for (let c = 0.1; c <= maxC + 1e-9; c += step) {
   527→   527→      pts.push({ cRate: +c.toFixed(2), usable: effectiveCapacityPct(design, +c.toFixed(2)) });
   528→   528→    }
   529→   529→    pts.push({ cRate: +maxC.toFixed(2), usable: effectiveCapacityPct(design, maxC) });
   530→   530→    return pts;
   531→   531→  }, [design, ready, maxC]);
   532→   532→
   533→   533→  // voltage sag at 1C: pure ohmic drop I·R0 (1C current = capacity in amps)
   534→   534→  const sag1C = ready ? design.capacityAh * design.r0 * 1000 : 0;
   535→   535→  const heatMax = jouleHeatW(design, maxC);
   536→   536→  const dT30 = tempRiseAdiabatic(design, maxC, 1800);
   537→   537→  const runtime1C = ready ? +(60 * (effectiveCapacityPct(design, 1) ?? 0) / 100).toFixed(0) : null;
   538→   538→  const usableAtMax = ready ? effectiveCapacityPct(design, maxC) : null;
   539→   539→
   540→   540→  if (!ready) {
   541→   541→    return (
   542→   542→      <Card>
   543→   543→        <CardHeader className="pb-2">
   544→   544→          <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-primary" /> Real-World Preview</CardTitle>
   545→   545→          <CardDescription>Enter capacity, R₀ and the voltage window to activate the preview.</CardDescription>
   546→   546→        </CardHeader>
   547→   547→      </Card>
   548→   548→    );
   549→   549→  }
   550→   550→
   551→   551→  return (
   552→   552→    <Card>
   553→   553→      <CardHeader className="pb-2">
   554→   554→        <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-primary" /> Real-World Preview</CardTitle>
   555→   555→        <CardDescription>
   556→   556→          Live from your values — ohmic estimate; full electro-thermal runs in Simulation.
   557→   557→        </CardDescription>
   558→   558→      </CardHeader>
   559→   559→      <CardContent className="space-y-4">
   560→   560→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   561→   561→          <StatCard
   562→   562→            label="Sag @1C (ohmic)"
   563→   563→            value={sag1C.toFixed(0)} unit="mV"
   564→   564→            hint="V drop below OCV from I·R₀"
   565→   565→          />
   566→   566→          <StatCard
   567→   567→            label={`Heat @ ${maxC}C`}
   568→   568→            value={heatMax ?? "—"} unit={heatMax != null ? "W" : undefined}
   569→   569→            hint="I²·R₀ joule heating"
   570→   570→            tone={heatMax != null && heatMax > 5 ? "warn" : "default"}
   571→   571→          />
   572→   572→          <StatCard
   573→   573→            label={`ΔT @ ${maxC}C, 30 min`}
   574→   574→            value={dT30 ?? "—"} unit={dT30 != null ? "°C" : undefined}
   575→   575→            hint="adiabatic estimate"
   576→   576→            tone={dT30 != null && dT30 > 25 ? "fail" : dT30 != null && dT30 > 10 ? "warn" : "default"}
   577→   577→          />
   578→   578→          <StatCard
   579→   579→            label="Runtime @ 1C → Vmin"
   580→   580→            value={runtime1C != null && runtime1C > 0 ? runtime1C : "—"} unit={runtime1C != null && runtime1C > 0 ? "min" : undefined}
   581→   581→            hint="full charge to cutoff"
   582→   582→          />
