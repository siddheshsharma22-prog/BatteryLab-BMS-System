    94→export function DesignTab() {
    95→  const { design, updateDesign, resetDesign, pack, setPack, sendCopilotMessage, copilotBusy } = useLabStore();
    96→  const { toast } = useToast();
    97→  const [saved, setSaved] = useState<SavedDesign[]>([]);
    98→  const [saving, setSaving] = useState(false);
    99→  const [pickId, setPickId] = useState<string>("");
   100→
   101→  const loadSaved = async () => {
   102→    try {
   103→      const res = await fetch("/api/designs");
   104→      const data = await res.json();
   105→      if (res.ok) setSaved(data.designs);
   106→    } catch {
   107→      /* silent */
   108→    }
   109→  };
   110→
   111→  useEffect(() => {
   112→    loadSaved();
   113→  }, []);
   114→
   115→  const issues = designIssues(design);
   116→  const complete = issues.length === 0;
   117→  const pm = packMath(design, pack);
   118→  const cellWh = design.capacityAh * (design.nominalV > 0 ? design.nominalV : (design.vmax + design.vmin) / 2 + 0.05);
   119→  const whKg = design.massKg > 0 ? cellWh / design.massKg : 0;
   120→
   121→  // Pack targets → S×P. Applied live: the Connection tab receives the cell
   122→  // count derived from the design the user just set.
   123→  const cellNominalV = design.nominalV > 0
   124→    ? design.nominalV
   125→    : design.vmax > 0 && design.vmin > 0 ? (design.vmax + design.vmin) / 2 + 0.05 : 0;
   126→  const sugS = pack.targetV > 0 && cellNominalV > 0 ? Math.max(1, Math.round(pack.targetV / cellNominalV)) : null;
   127→  const sugP = pack.targetAh > 0 && design.capacityAh > 0 ? Math.max(1, Math.ceil(pack.targetAh / design.capacityAh)) : null;
   128→  const setTarget = (patch: { targetV?: number; targetAh?: number }) => {
   129→    const tv = patch.targetV ?? pack.targetV;
   130→    const ta = patch.targetAh ?? pack.targetAh;
   131→    const s = tv > 0 && cellNominalV > 0 ? Math.max(1, Math.round(tv / cellNominalV)) : null;
   132→    const p = ta > 0 && design.capacityAh > 0 ? Math.max(1, Math.ceil(ta / design.capacityAh)) : null;
   133→    setPack({ ...patch, ...(s ? { series: s } : {}), ...(p ? { parallel: p } : {}) });
   134→  };
   135→
   136→  const save = async () => {
   137→    if (!complete) {
   138→      toast({ title: "Design incomplete", description: "Fill every required field before saving.", variant: "destructive" });
   139→      return;
   140→    }
   141→    setSaving(true);
   142→    try {
   143→      const res = await fetch("/api/designs", {
   144→        method: "POST",
   145→        headers: { "Content-Type": "application/json" },
   146→        body: JSON.stringify({ ...design, ...pack }),
   147→      });
   148→      const data = await res.json();
   149→      if (!res.ok) throw new Error(data.error);
   150→      toast({ title: "Design saved", description: `${design.name} is stored in the database.` });
   151→      loadSaved();
   152→    } catch (e) {
   153→      toast({ title: "Save failed", description: String(e instanceof Error ? e.message : e), variant: "destructive" });
   154→    } finally {
   155→      setSaving(false);
   156→    }
   157→  };
   158→
   159→  const load = (d: SavedDesign) => {
   160→    const n = (v: number | null | undefined, dflt = 0) => (v === null || v === undefined ? dflt : v);
   161→    const s = (v: string | null) => v ?? "";
   162→    const loaded: Partial<CellDesign> = {
   163→      name: d.name, chemistry: d.chemistry as CellDesign["chemistry"],
   164→      brand: s(d.brand), model: s(d.model),
   165→      formFactor: s(d.formFactor), cathode: s(d.cathode), anode: s(d.anode),
   166→      separator: s(d.separator), electrolyte: s(d.electrolyte), application: s(d.application),
   167→      sourceRef: s(d.sourceRef),
   168→      capacityAh: d.capacityAh, nominalV: n(d.nominalV), vmax: d.vmax, vmin: d.vmin, r0: d.r0,
   169→      r1: n(d.r1), c1: n(d.c1), r2: n(d.r2), c2: n(d.c2),
   170→      maxDischargeC: n(d.maxDischargeC), maxPulseDischargeC: n(d.maxPulseDischargeC),
   171→      maxChargeC: n(d.maxChargeC), stdChargeC: n(d.stdChargeC),
   172→      diameterMm: n(d.diameterMm), lengthMm: n(d.lengthMm), widthMm: n(d.widthMm),
   173→      heightMm: n(d.heightMm), thicknessMm: n(d.thicknessMm), massKg: d.massKg,
   174→      cpJkgK: n(d.cpJkgK), areaM2: n(d.areaM2), entropicVperK: n(d.entropicVperK, 0.0001),
   175→      cycleLifeCycles: n(d.cycleLifeCycles), cycleLifeNote: s(d.cycleLifeNote),
   176→      chargeTempMinC: n(d.chargeTempMinC), chargeTempMaxC: n(d.chargeTempMaxC),
   177→      dischargeTempMinC: n(d.dischargeTempMinC), dischargeTempMaxC: n(d.dischargeTempMaxC),
   178→      selfDischargePctMonth: n(d.selfDischargePctMonth),
   179→    };
   180→    updateDesign(loaded);
   181→    // Saved designs carry their cell count (and no pack targets) — restore
   182→    // S/P from the DB and clear any targets left over from a previous session.
   183→    useLabStore.getState().setPack({ series: d.seriesCount, parallel: d.parallelCount, targetV: 0, targetAh: 0 });
   184→    toast({ title: "Design loaded", description: `${d.name} — all saved values restored.` });
   185→  };
   186→
   187→  const remove = async (id: string) => {
   188→    try {
   189→      await fetch(`/api/designs/${id}`, { method: "DELETE" });
   190→      loadSaved();
   191→      toast({ title: "Design deleted" });
   192→    } catch {
   193→      toast({ title: "Delete failed", variant: "destructive" });
   194→    }
   195→  };
   196→
   197→  const askAi = (prompt: string) => sendCopilotMessage(prompt);
   198→
   199→  const pickCell = (id: string) => {
   200→    setPickId(id);
   201→    const cell = getLibraryCell(id);
   202→    if (!cell) return;
   203→    useLabStore.getState().loadLibraryCell(cell);
   204→    toast({
   205→      title: "Datasheet loaded",
   206→      description: `${cell.brand} ${cell.model} — every value below is now editable.`,
   207→    });
   208→  };
   209→
   210→  return (
   211→    <div className="space-y-6">
   212→      <SectionTitle
   213→        icon={<BatteryCharging className="h-5 w-5" />}
   214→        title="1 · Battery Design"
   215→        desc="Every input needed to define your cell — start from a real datasheet or a blank sheet."
   216→      />
   217→
   218→      {/* Starting point: blank or from datasheet */}
   219→      <Card className="border-primary/30 bg-primary/[0.03]">
   220→        <CardContent className="p-4 flex flex-col lg:flex-row lg:items-center gap-3">
   221→          <div className="flex-1">
   222→            <div className="text-sm font-medium flex items-center gap-2">
   223→              <BookOpen className="h-4 w-4 text-primary" /> Starting point
   224→            </div>
   225→            <p className="text-xs text-muted-foreground mt-0.5">
   226→              Pre-fill every field from {LIBRARY_CELLS.length} real datasheets, or start blank — loaded values stay editable.
   227→            </p>
   228→          </div>
   229→          <div className="flex flex-col sm:flex-row gap-2 lg:w-[520px]">
   230→            <Select value={pickId} onValueChange={pickCell}>
   231→              <SelectTrigger className="h-9 flex-1" aria-label="Pick a datasheet as starting point">
   232→                <SelectValue placeholder="Load from datasheet…" />
   233→              </SelectTrigger>
   234→              <SelectContent>
   235→                {LIBRARY_CATEGORIES.map((cat) => {
   236→                  const cells = LIBRARY_CELLS.filter((c) => c.category === cat);
   237→                  if (cells.length === 0) return null;
   238→                  return (
   239→                    <SelectGroup key={cat}>
   240→                      <SelectLabel>{cat}</SelectLabel>
   241→                      {cells.map((c: LibraryCell) => (
   242→                        <SelectItem key={c.id} value={c.id}>
   243→                          {c.brand} — {c.model} ({c.capacityAh} Ah / {c.nominalV} V)
   244→                        </SelectItem>
   245→                      ))}
   246→                    </SelectGroup>
   247→                  );
   248→                })}
   249→              </SelectContent>
   250→            </Select>
   251→            <Button variant="outline" onClick={() => { resetDesign(); setPickId(""); toast({ title: "Design reset to blank sheet" }); }}>
   252→              <RotateCcw className="h-3.5 w-3.5 mr-1" /> Blank sheet
   253→            </Button>
   254→          </div>
   255→        </CardContent>
   256→      </Card>
   257→
   258→      {/* Completeness checklist */}
   259→      <Card className={complete ? "border-primary/40 bg-primary/[0.04]" : ""}>
   260→        <CardContent className="p-4">
   261→          <div className="flex items-center justify-between gap-3">
   262→            <div className="text-sm font-medium flex items-center gap-2">
   263→              {complete ? (
   264→                <><CheckCircle2 className="h-4 w-4 text-primary" /> Core definition complete — next: Connection, then Simulation</>
   265→              ) : (
   266→                <><CircleDashed className="h-4 w-4 text-amber-600" /> Design checklist — {issues.length} to go</>
   267→              )}
   268→            </div>
   269→            {design.sourceRef ? (
   270→              <Badge variant="outline" className="text-[10px] max-w-[280px] truncate">{design.sourceRef}</Badge>
   271→            ) : null}
   272→          </div>
   273→          {!complete && (
   274→            <ul className="grid grid-cols-1 sm:grid-cols-2 gap-x-4 gap-y-1 mt-2">
   275→              {issues.map((i) => (
   276→                <li key={i} className="text-xs text-muted-foreground flex items-center gap-1.5">
   277→                  <span className="h-1.5 w-1.5 rounded-full bg-amber-500 shrink-0" /> {i}
   278→                </li>
   279→              ))}
   280→            </ul>
   281→          )}
   282→        </CardContent>
   283→      </Card>
   284→      <div className="flex flex-wrap gap-2">
   285→        <button
   286→          onClick={() => askAi("I'm starting from a blank sheet. Guide me step by step — first ask me what I'm building.")}
   287→          disabled={copilotBusy}
   288→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   289→        >
   290→          <Sparkles className="h-3.5 w-3.5" /> Guide me from zero
   291→        </button>
   292→        <button
   293→          onClick={() => askAi("Recommend a real cell from the Cell Library for my application and explain why, with the apply block so I can load it.")}
   294→          disabled={copilotBusy}
   295→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   296→        >
   297→          <Sparkles className="h-3.5 w-3.5" /> Recommend a real cell
   298→        </button>
   299→        <button
   300→          onClick={() => askAi("Suggest a complete cell parameter set for my application (I'll describe it next), including the apply block so I can load it with one click.")}
   301→          disabled={copilotBusy}
   302→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   303→        >
   304→          <Sparkles className="h-3.5 w-3.5" /> Suggest values for my application
   305→        </button>
   306→        <button
   307→          onClick={() => askAi("Review my current cell parameters and warn me about anything unsafe or unrealistic.")}
   308→          disabled={copilotBusy}
   309→          className="inline-flex items-center gap-1.5 rounded-md border bg-card px-2.5 py-1.5 text-xs font-medium text-foreground hover:bg-muted transition-colors disabled:opacity-50"
   310→        >
   311→          <Sparkles className="h-3.5 w-3.5" /> Review my numbers
   312→        </button>
   313→      </div>
   314→
   315→      {/* ===== Input sections ===== */}
   316→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   317→        {/* Identity & materials */}
   318→        <Card>
   319→          <CardHeader className="pb-2">
   320→            <CardTitle className="text-sm flex items-center gap-2"><BatteryCharging className="h-4 w-4 text-primary" /> Identity &amp; materials</CardTitle>
   321→            <CardDescription>What the battery is and what it&apos;s made of</CardDescription>
   322→          </CardHeader>
   323→          <CardContent className="grid grid-cols-2 gap-3">
   324→            <div className="col-span-2">
   325→              <Label htmlFor="dname">Design name</Label>
   326→              <Input id="dname" className="mt-1" placeholder="e.g. My 21700 traction cell" value={design.name} onChange={(e) => updateDesign({ name: e.target.value })} />
   327→            </div>
   328→            <div>
   329→              <TextField label="Brand (based on)" value={design.brand} placeholder="e.g. Samsung SDI" onChange={(v) => updateDesign({ brand: v })} />
   330→            </div>
   331→            <div>
   332→              <TextField label="Model (based on)" value={design.model} placeholder="e.g. INR21700-40T" onChange={(v) => updateDesign({ model: v })} />
   333→            </div>
   334→            <div className="col-span-2">
   335→              <Label className="text-xs text-muted-foreground">Chemistry (OCV curve family)</Label>
   336→              <Select value={design.chemistry} onValueChange={(v) => updateDesign({ chemistry: v as Chemistry })}>
   337→                <SelectTrigger className="mt-1 h-9"><SelectValue /></SelectTrigger>
   338→                <SelectContent>
   339→                  {CHEM_OPTIONS.map((c) => (
   340→                    <SelectItem key={c.id} value={c.id}>
   341→                      <span className="font-medium">{c.label}</span>
   342→                      <span className="text-muted-foreground text-xs"> — {c.hint}</span>
   343→                    </SelectItem>
   344→                  ))}
   345→                </SelectContent>
   346→              </Select>
   347→            </div>
   348→            <div className="col-span-2">
   349→              <Label className="text-xs text-muted-foreground">Form factor</Label>
   350→              <Select value={design.formFactor} onValueChange={(v) => updateDesign({ formFactor: v })}>
   351→                <SelectTrigger className="mt-1 h-9"><SelectValue placeholder="Select or leave custom" /></SelectTrigger>
   352→                <SelectContent>
   353→                  {FORM_FACTORS.map((f) => (
   354→                    <SelectItem key={f} value={f}>{f}</SelectItem>
   355→                  ))}
   356→                </SelectContent>
   357→              </Select>
   358→            </div>
   359→            <TextField label="Cathode (+) material" value={design.cathode} placeholder="e.g. NMC 811, LiFePO₄" onChange={(v) => updateDesign({ cathode: v })} />
   360→            <TextField label="Anode (−) material" value={design.anode} placeholder="e.g. Graphite + SiOx" onChange={(v) => updateDesign({ anode: v })} />
   361→            <TextField label="Separator" value={design.separator} placeholder="e.g. PE/PP microporous" onChange={(v) => updateDesign({ separator: v })} />
   362→            <TextField label="Electrolyte" value={design.electrolyte} placeholder="e.g. LiPF₆ carbonate" onChange={(v) => updateDesign({ electrolyte: v })} />
   363→            <div className="col-span-2">
   364→              <TextField label="Target application" value={design.application} placeholder="e.g. e-bike pack, home storage, drone" onChange={(v) => updateDesign({ application: v })} />
   365→            </div>
   366→          </CardContent>
   367→        </Card>
   368→
   369→        {/* Geometry & mass */}
   370→        <Card>
   371→          <CardHeader className="pb-2">
   372→            <CardTitle className="text-sm flex items-center gap-2"><Ruler className="h-4 w-4 text-primary" /> Format, geometry &amp; mass</CardTitle>
   373→            <CardDescription>Physical envelope — also used to estimate surface area for thermal testing</CardDescription>
   374→          </CardHeader>
   375→          <CardContent className="grid grid-cols-2 gap-3">
   376→            <NumField label="Diameter Ø (mm)" value={design.diameterMm} step={0.05} min={0} onChange={(v) => updateDesign({ diameterMm: v })} hint="cylindrical cells" />
   377→            <NumField label="Length / height (mm)" value={design.lengthMm} step={0.1} min={0} onChange={(v) => updateDesign({ lengthMm: v })} hint="cylinder length" />
   378→            <NumField label="Width (mm)" value={design.widthMm} step={0.1} min={0} onChange={(v) => updateDesign({ widthMm: v })} hint="prismatic" />
   379→            <NumField label="Height (mm)" value={design.heightMm} step={0.1} min={0} onChange={(v) => updateDesign({ heightMm: v })} hint="prismatic" />
   380→            <NumField label="Thickness (mm)" value={design.thicknessMm} step={0.1} min={0} onChange={(v) => updateDesign({ thicknessMm: v })} hint="pouch" />
   381→            <NumField label="Mass (g)" value={design.massKg * 1000} step={0.1} min={0} onChange={(v) => updateDesign({ massKg: v / 1000 })} hint={`${(design.massKg * 1000).toFixed(1)} g total`} />
   382→            <div className="col-span-2 flex items-center justify-between gap-2 rounded-lg bg-muted/50 px-3 py-2">
   383→              <p className="text-xs text-muted-foreground">
   384→                Estimated from dimensions: {estimateAreaM2(design) > 0 ? <span className="font-medium text-foreground">{(estimateAreaM2(design) * 10000).toFixed(0)} cm²</span> : "add dimensions"}
   385→              </p>
   386→              <Button
   387→                size="sm" variant="outline"
   388→                disabled={estimateAreaM2(design) <= 0}
   389→                onClick={() => {
   390→                  const a = estimateAreaM2(design);
   391→                  if (a > 0) { updateDesign({ areaM2: a }); toast({ title: "Surface area estimated", description: `${(a * 10000).toFixed(0)} cm² from your dimensions.` }); }
   392→                }}
   393→              >
   394→                <Ruler className="h-3.5 w-3.5 mr-1" /> Apply estimate
   395→              </Button>
   396→            </div>
   397→          </CardContent>
   398→        </Card>
   399→
   400→        {/* Electrical ratings */}
   401→        <Card>
   402→          <CardHeader className="pb-2">
   403→            <CardTitle className="text-sm flex items-center gap-2"><Gauge className="h-4 w-4 text-primary" /> Electrical ratings</CardTitle>
   404→            <CardDescription>Everything the datasheet front page lists</CardDescription>
   405→          </CardHeader>
   406→          <CardContent className="grid grid-cols-2 gap-3">
   407→            <NumField label="Capacity (Ah)" value={design.capacityAh} step={0.1} min={0} onChange={(v) => updateDesign({ capacityAh: v })} />

   557→// Real-world condition preview — computed live from the

   560→
   561→function RealWorldPreview() {
   562→  const design = useLabStore((s) => s.design);
   563→  const ready = design.capacityAh > 0 && design.r0 > 0 && design.vmin > 0 && design.vmax > design.vmin;
   564→  const maxC = Math.min(design.maxDischargeC > 0 ? design.maxDischargeC : 3, 12);
   565→
   566→  // Loaded voltage vs SOC at several C-rates
   567→  const sagData = useMemo(() => {
   568→    if (!ready) return [];
   569→    const rates = [0.25, 0.5, 1, 2, maxC].filter((r, i, arr) => arr.indexOf(r) === i).sort((a, b) => a - b);
   570→    const pts: { soc: number; [k: string]: number }[] = [];
   571→    for (let soc = 100; soc >= 0; soc -= 2) {
   572→      const row: { soc: number; [k: string]: number } = { soc };
   573→      for (const r of rates) row[`${r}C`] = +loadedVoltageAtSoc(design, soc / 100, r).toFixed(3);
   574→      pts.push(row);
   575→    }
   576→    return pts;
   577→  }, [design, ready, maxC]);
   578→
   579→  // Effective capacity vs C-rate
   580→  const effData = useMemo(() => {
   581→    if (!ready) return [];
   582→    const pts: { cRate: number; usable: number | null }[] = [];
   583→    const step = maxC <= 2 ? 0.1 : 0.25;
   584→    for (let c = 0.1; c <= maxC + 1e-9; c += step) {
   585→      pts.push({ cRate: +c.toFixed(2), usable: effectiveCapacityPct(design, +c.toFixed(2)) });
   586→    }
   587→    pts.push({ cRate: +maxC.toFixed(2), usable: effectiveCapacityPct(design, maxC) });
   588→    return pts;
   589→  }, [design, ready, maxC]);
   590→
   591→  // voltage sag at 1C: pure ohmic drop I·R0 (1C current = capacity in amps)
   592→  const sag1C = ready ? design.capacityAh * design.r0 * 1000 : 0;
   593→  const heatMax = jouleHeatW(design, maxC);
   594→  const dT30 = tempRiseAdiabatic(design, maxC, 1800);
   595→  const runtime1C = ready ? +(60 * (effectiveCapacityPct(design, 1) ?? 0) / 100).toFixed(0) : null;
   596→  const usableAtMax = ready ? effectiveCapacityPct(design, maxC) : null;
   597→
   598→  if (!ready) {
   599→    return (
   600→      <Card>
   601→        <CardHeader className="pb-2">
   602→          <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-primary" /> Real-World Preview</CardTitle>
   603→          <CardDescription>Enter capacity, R₀ and the voltage window to activate the preview.</CardDescription>
   604→        </CardHeader>
   605→      </Card>
   606→    );
   607→  }
   608→
   609→  return (
   610→    <Card>
   611→      <CardHeader className="pb-2">
   612→        <CardTitle className="text-sm flex items-center gap-2"><Activity className="h-4 w-4 text-primary" /> Real-World Preview</CardTitle>
   613→        <CardDescription>
   614→          Live from your values — ohmic estimate; full electro-thermal runs in Simulation.
   615→        </CardDescription>
   616→      </CardHeader>
   617→      <CardContent className="space-y-4">
   618→        <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   619→          <StatCard
   620→            label="Sag @1C (ohmic)"
   621→            value={sag1C.toFixed(0)} unit="mV"
   622→            hint="V drop below OCV from I·R₀"
   623→          />
   624→          <StatCard
   625→            label={`Heat @ ${maxC}C`}
   626→            value={heatMax ?? "—"} unit={heatMax != null ? "W" : undefined}
   627→            hint="I²·R₀ joule heating"
   628→            tone={heatMax != null && heatMax > 5 ? "warn" : "default"}
   629→          />
   630→          <StatCard
   631→            label={`ΔT @ ${maxC}C, 30 min`}
   632→            value={dT30 ?? "—"} unit={dT30 != null ? "°C" : undefined}
   633→            hint="adiabatic estimate"
   634→            tone={dT30 != null && dT30 > 25 ? "fail" : dT30 != null && dT30 > 10 ? "warn" : "default"}
   635→          />
   636→          <StatCard
   637→            label="Runtime @ 1C → Vmin"
   638→            value={runtime1C != null && runtime1C > 0 ? runtime1C : "—"} unit={runtime1C != null && runtime1C > 0 ? "min" : undefined}
   639→            hint="full charge to cutoff"
   640→          />
   641→        </div>
   642→
   643→        <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   644→          <ChartCard title="Loaded voltage vs SOC — voltage sag under real loads">
   645→            <ResponsiveContainer width="100%" height="100%">
   646→              <LineChart data={sagData} margin={{ top: 8, right: 16, bottom: 8, left: 0 }}>
   647→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
   648→                <XAxis dataKey="soc" tick={{ fontSize: 11 }} unit="%" />
   649→                <YAxis domain={["auto", "auto"]} tick={{ fontSize: 11 }} unit=" V" />
   650→                <Tooltip formatter={(v: number | string) => [`${v} V`, ""]} labelFormatter={(l) => `SOC ${l}%`} />
   651→                <Legend wrapperStyle={{ fontSize: 11 }} />
   652→                <ReferenceLine y={design.vmin} stroke={CHART_COLORS.red} strokeDasharray="4 4" label={{ value: "Vmin", fontSize: 10, fill: CHART_COLORS.red }} />
   653→                {[0.25, 0.5, 1, 2, maxC].filter((r, i, a) => a.indexOf(r) === i).map((r, i) => (
   654→                  <Line
   655→                    key={r}
   656→                    type="monotone" dataKey={`${r}C`} name={`${r}C (${(r * design.capacityAh).toFixed(1)} A)`}
   657→                    stroke={[CHART_COLORS.blue, CHART_COLORS.cyan, CHART_COLORS.slate, CHART_COLORS.amber, CHART_COLORS.red][i % 5]}
   658→                    strokeWidth={2} dot={false}
   659→                  />
   660→                ))}
   661→              </LineChart>
   662→            </ResponsiveContainer>
   663→          </ChartCard>
   664→
   665→          <ChartCard title="Usable capacity vs C-rate — real-world derating">
   666→            <ResponsiveContainer width="100%" height="100%">
   667→              <LineChart data={effData} margin={{ top: 8, right: 16, bottom: 8, left: 0 }}>
   668→                <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" />
   669→                <XAxis dataKey="cRate" tick={{ fontSize: 11 }} unit="C" />
   670→                <YAxis domain={[0, 100]} tick={{ fontSize: 11 }} unit="%" />
   671→                <Tooltip formatter={(v: number | string) => [`${v} %`, "usable"]} labelFormatter={(l) => `${l}C`} />
   672→                <Legend wrapperStyle={{ fontSize: 11 }} />
   673→                <Line type="monotone" dataKey="usable" name="Usable capacity" stroke={CHART_COLORS.cyan} strokeWidth={2.5} dot={false} />
   674→              </LineChart>
   675→            </ResponsiveContainer>
   676→          </ChartCard>
   677→        </div>
   678→
   679→        <div className="flex items-center gap-2 text-xs text-muted-foreground">
   680→          <TrendingDown className="h-3.5 w-3.5 text-primary shrink-0" />
   681→          <span>≈<span className="font-semibold text-foreground">{usableAtMax ?? "—"}%</span> of rated capacity is usable at {maxC}C before the loaded voltage hits Vmin.</span>
   682→        </div>
   683→      </CardContent>
   684→    </Card>
   685→  );
   686→}

   692→function NumField({ label, value, step, min, onChange, hint, placeholder }: { label: string; value: number; step: number; min?: number; onChange: (v: number) => void; hint?: string; placeholder?: string }) {
   693→  return (
   694→    <div>
   695→      <Label className="text-xs text-muted-foreground">{label}</Label>
   696→      <Input
   697→        className="mt-1 h-8"
   698→        type="number"
   699→        value={value || ""}
   700→        placeholder={placeholder ?? "0"}
   701→        step={step}
   702→        min={min}
   703→        onChange={(e) => {
   704→          const v = parseFloat(e.target.value);
   705→          if (!isNaN(v)) onChange(v);
   706→          else if (e.target.value === "") onChange(0);
   707→        }}
   708→      />
   709→      {hint && <p className="text-[10px] text-muted-foreground mt-0.5">{hint}</p>}
   710→    </div>
   711→  );
   712→}
   713→
   714→function TextField({ label, value, placeholder, onChange }: { label: string; value: string; placeholder?: string; onChange: (v: string) => void }) {
   715→  return (
   716→    <div>
   717→      <Label className="text-xs text-muted-foreground">{label}</Label>
   718→      <Input
   719→        className="mt-1 h-8"
   720→        value={value}
   721→        placeholder={placeholder}
   722→        onChange={(e) => onChange(e.target.value)}
   723→      />
   724→    </div>
   725→  );
   726→}
